const dummy = [
  { name: "總量", type: '', count: app.jsonData_.length  },
  { name: "漲", type: '',denominator:"總量", count: app.jsonData_.filter(x=> x.rt_price>=0).length },
  { name: "跌", type: '',denominator:"總量", count: app.jsonData_.filter(x=> x.rt_price<0).length },
  
];

//  總價差 
//  rt_量

// 生成 ratio
const withRatio = dummy.map(item => {
  if (item.denominator) {
    // 於 dummy array中找到 denominator對應的項目（以name屬性），並取得其count
    const denomObj = dummy.find(d => d.name === item.denominator);
    const denomCount = denomObj ? denomObj.count : undefined;
    return {
      ...item,
      ratio: parseFloat(((denomCount ? item.count / denomCount : undefined) *100 ).toFixed(1))
    };
  } else {
    return item;
  }
});
// 顯示結果
console.table(withRatio );

function getFlag(rt_price) {
  if (rt_price > 3) return 1;
  if (rt_price > 2 && rt_price <= 3) return 2;
  if (rt_price >= 0 && rt_price <= 2) return 3;
  return 0; // 其他（如負數），可依需求顯示
}

// 指定群組欄位名稱
const groupKey = 'Typelevel'; // 可改成你想分組的欄位，例如 'Type2', 'Type1'...

const result = [];
const typeMap  = {};

app.jsonData_.forEach(item => {
  const key = item[groupKey]; // 動態取欄位值
  if (!typeMap[key]) {
    typeMap[key] = {
      [groupKey]: key,
      totalCount: 0,
      rt_price_positive_count: 0,
      rt_price_negative_count: 0
    };
    result.push(typeMap[key]);
  }
  typeMap[key].totalCount += 1;
  if (item.rt_price >= 0) typeMap[key].rt_price_positive_count += 1;
  if (item.rt_price < 0) typeMap[key].rt_price_negative_count += 1;
});

const typeArray = Object.values(typeMap);
// 若要按分組欄位排序
typeArray.sort((a, b) => String(a[groupKey]).localeCompare(String(b[groupKey])));

console.table(typeArray);
